cmake_minimum_required(VERSION 3.26)
project(roce_megamoe_probe LANGUAGES CXX CUDA)
find_package(CUDAToolkit REQUIRED)
find_path(MSCCLPP_INCLUDE_DIR mscclpp/port_channel.hpp REQUIRED)
find_library(MSCCLPP_LIBRARY NAMES mscclpp REQUIRED)
add_executable(gpunetio_payload gpunetio_payload.cu)
target_compile_features(gpunetio_payload PRIVATE cxx_std_17 cuda_std_17)
target_include_directories(gpunetio_payload PRIVATE ${MSCCLPP_INCLUDE_DIR})
target_link_libraries(gpunetio_payload PRIVATE ${MSCCLPP_LIBRARY} CUDA::cudart)
